#!/bin/sh set -eu tmp="$(mktemp -d)" trap 'rm -rf "$tmp"' EXIT HUP INT TERM curl -fsSL \ https://github.com/mkvlrn/arch-setup/releases/latest/download/arch-setup \ -o "$tmp/arch-setup" chmod +x "$tmp/arch-setup" "$tmp/arch-setup" printf '\nRestoring machine secrets\n\n' age -d "$HOME/repos/arch-setup/secrets/secrets.tar.age" | tar -C "$HOME" -xf -
