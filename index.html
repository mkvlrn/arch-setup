#!/bin/sh set -eu repo="$HOME/repos/arch-setup" if [ -e "$repo" ]; then printf 'Repository path already exists: %s\n' "$repo" >&2 exit 1 fi if ! command -v git >/dev/null 2>&1; then sudo pacman -Sy --noconfirm git fi mkdir -p "$(dirname "$repo")" git clone https://github.com/mkvlrn/arch-setup "$repo" chmod +x "$repo/install.sh" SETUP_REPO_DIR="$repo" "$repo/install.sh" printf '\nRestoring machine secrets\n\n' age -d "$HOME/repos/arch-setup/secrets/secrets.tar.age" | tar -C "$HOME" -xf -
